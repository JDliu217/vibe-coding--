import assert from 'node:assert/strict';
import {chromium} from '/workspace/.onboarding-tools/node_modules/playwright/index.mjs';
const root='/workspace/.review-skills/consumer-73abc72';
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox']});
const results=[];
const reviewPhone='139'+String(Date.now()).slice(-8);
async function check(name,fn){try{results.push({name,status:'PASS',evidence:await fn()});}catch(e){results.push({name,status:'FAIL',error:e.message});}}
async function login(page,phone,password){
  await page.getByPlaceholder('手机号',{exact:true}).fill(phone);
  await page.getByPlaceholder('密码',{exact:true}).fill(password);
  await page.getByRole('button',{name:'登录',exact:true}).click();
  await page.waitForURL(/\/(products|cart)(?:\?|$)/);
}
await check('registered user buys a product, views order and cancels through real services',async()=>{
  const context=await browser.newContext({viewport:{width:1365,height:900}});const page=await context.newPage();
  const errors=[];page.on('pageerror',e=>errors.push(e.message));
  await page.goto('http://127.0.0.1:3000/register');
  await page.getByPlaceholder('手机号',{exact:true}).fill(reviewPhone);
  await page.getByPlaceholder('昵称，可选').fill('浏览器验收');
  await page.getByPlaceholder('密码至少 6 位').fill('Review123456');
  await page.getByRole('button',{name:'注册',exact:true}).click();
  await page.waitForURL('**/login');
  await login(page,reviewPhone,'Review123456');
  await page.locator('article').first().waitFor();
  await page.screenshot({path:root+'/products.png',fullPage:true});
  await page.locator('article').filter({has:page.getByRole('link',{name:'香蕉',exact:true})}).getByRole('button',{name:'加入购物车'}).click();
  await page.getByRole('link',{name:/^购物车/}).click();
  await page.getByRole('button',{name:'提交订单'}).click();
  await page.waitForURL(/\/orders\/\d+$/);
  await page.getByText('状态：已创建',{exact:true}).waitFor();
  const orderPath=new URL(page.url()).pathname;
  await page.screenshot({path:root+'/order.png',fullPage:true});
  await page.getByRole('button',{name:'取消订单',exact:true}).click();
  await page.getByText('状态：已取消',{exact:true}).waitFor();
  assert.deepEqual(errors,[]);
  await context.close();return {orderPath,status:'cancelled',browserExceptions:0};
});
